// negH - the half that was missing: the Sales Invoice screen.
//
// The company setting and the API existed; the invoice form had no way to
// use either, so with the setting on a user still could not post the
// invoice. That is not a feature, it is a promise. Run negG first - this
// reads the company setting that screen sets.
//
// NOBODY IS OFFERED THE OVERRIDE UP FRONT. The invoice is posted normally.
// Only when the server refuses it for want of stock, AND the company
// permits the override, does a panel appear carrying the server's own
// message - 'Only 4 in stock at this branch' - with a box for the reason
// and a Post anyway button. Then the same invoice goes again with the
// override attached.
//
// Asking afterwards rather than beforehand is the design, not a shortcut.
// A tick-box on an empty form gets ticked out of habit by somebody who has
// not been told anything is wrong yet. A panel that appears saying how
// short you are is a decision taken in front of the fact - and it carries
// the two consequences with it, because somebody choosing this is taking
// on an estimated margin and a negative stock balance.
//
// A FIFO refusal is deliberately NOT matched: the message says 'not under
// FIFO', there is nothing the user could usefully confirm, and it shows as
// an ordinary error.
//
// negI adds the panel itself. Run both; between them the screen compiles
// but the override is unreachable.
//
// Save this as backend/tests/negH.mjs and run it from backend/:
//   node tests/negH.mjs
// Safe to run twice - a second run says 'already there' and changes nothing.

import fs from "fs";
import path from "path";
import crypto from "crypto";
import { fileURLToPath } from "url";

const here = path.dirname(fileURLToPath(import.meta.url));
const at = (f) => path.join(here, "..", f);
const read = (f) => fs.readFileSync(at(f), "utf8").replace(/\r\n/g, "\n");
const L = (...ls) => ls.join("\n");
const save = (f, t) => fs.writeFileSync(at(f), t.replace(/\n*$/, "\n"));

let applied = 0, already = 0;
function edit(file, from, to, done) {
  const t = read(file);
  if (t.includes(done)) { already++; save(file, t); return; }
  const n = t.split(from).length - 1;
  if (n === 0) throw new Error("anchor not found in " + file + ": " + from.slice(0, 70));
  if (n > 1) throw new Error("anchor is not unique in " + file + ": " + from.slice(0, 70));
  save(file, t.replace(from, to));
  applied++;
}
